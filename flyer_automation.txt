name: Generar Flyers Automaticamente

on:
  schedule:
    - cron: '0 20 * * *' # 3 PM Hora Perú
  workflow_dispatch:
  push:
    branches: [ main ]

jobs:
  build:
    runs-on: ubuntu-latest
    timeout-minutes: 120 # [!] Aumentamos a 2 horas por si el PDF gigante demora en subir
    strategy:
      matrix:
        rango: [
          {i: 0, f: 25}, {i: 25, f: 50}, {i: 50, f: 75}, {i: 75, f: 100},
          {i: 100, f: 125}, {i: 125, f: 150}, {i: 150, f: 175}, {i: 175, f: 200},
          {i: 200, f: 220}, {i: 220, f: 240}, 
          {i: 240, f: 245}, 
          {i: 245, f: 255}
        ]
      fail-fast: false

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Configurar Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'
          cache: 'pip'

      - name: Instalar librerias
        run: |
          pip install pandas requests Pillow gspread oauth2client google-api-python-client google-auth-oauthlib google-auth-httplib2 || sleep 10 && pip install pandas requests Pillow gspread oauth2client google-api-python-client google-auth-oauthlib google-auth-httplib2

      - name: Ejecutar Bloque (${{ matrix.rango.i }} - ${{ matrix.rango.f }})
        env:
          GOOGLE_SHEETS_JSON: ${{ secrets.GOOGLE_SHEETS_JSON }}
          GOOGLE_TOKEN_JSON: ${{ secrets.GOOGLE_TOKEN_JSON }}
          PYTHONUNBUFFERED: 1 
          PYTHONDONTWRITEBYTECODE: 1
        run: |
          # [!] AJUSTE DE ESPERA SENIOR (Factor 5)
          # Esto hace que las máquinas pesadas arranquen rápido y no se queden inactivas
          TOTAL_ESPERA=$(( (${{ matrix.rango.i }} / 25) * 5 ))
          echo ">>> [!] INICIO SEGURO: Espera de $TOTAL_ESPERA segundos..."
          
          CONTADOR=0
          while [ $CONTADOR -lt $TOTAL_ESPERA ]; do
            echo ">>> [STAY ALIVE] Preparando entorno masivo... ($((TOTAL_ESPERA - CONTADOR))s restantes)"
            sleep 30
            CONTADOR=$((CONTADOR + 30))
          done
          
          # [!] OPTIMIZACIÓN DE RED PARA PDFS GIGANTES (300MB+)
          git config --global http.postBuffer 1048576000
          git config --global core.compression 0
          
          # Limpieza absoluta de temporales antes de empezar
          rm -rf temp_flyers temp_cache_img
          mkdir -p temp_flyers temp_cache_img
          
          # [!] LÓGICA DE EJECUCIÓN CON CONTINGENCIA INTELIGENTE
          # Si el bloque es el 240 (el pesado), le damos un solo intento con toda la RAM disponible.
          # Para los demás, mantenemos el reintento.
          if [ ${{ matrix.rango.i }} -eq 240 ]; then
            echo ">>> [ZONA CRÍTICA] Procesando PDF Masivo..."
            python -u generar_flyers.py ${{ matrix.rango.i }} ${{ matrix.rango.f }}
          else
            python -u generar_flyers.py ${{ matrix.rango.i }} ${{ matrix.rango.f }} || (echo ">>> Reintentando por fallo leve..." && sleep 15 && python -u generar_flyers.py ${{ matrix.rango.i }} ${{ matrix.rango.f }})
          fi
